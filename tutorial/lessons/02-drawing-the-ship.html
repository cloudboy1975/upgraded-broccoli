Drawing the Ship

Now that the canvas is alive, let's put something on it — the ship, standing still for now (input arrives next lesson). This is also where state shows up: one plain object holding everything about the game world, read by every draw function. That single-object pattern is what the rest of the game is built on.

State as data

Rather than scattering separate variables for the ship's position, heading, and so on, everything lives in one state object. That pays off later: it's the one thing a restart has to reset, the one thing collision code reads from, and the one thing a test can inspect from the outside — which is exactly what the debug hook below is for.

JavaScript
var SHIP_RADIUS = 12;

var state = {
  ship: {
    heading: -Math.PI / 2, // pointing "up"
    radius: SHIP_RADIUS
  },
  canvasCssW: 0, canvasCssH: 0, canvasDpr: 1,
  timing: { lastFrameTime: 0 }
};
window.__astDebugState = state; // read-only hook used by this tutorial's own tests

Heading is in radians, and -Math.PI / 2 means "up" in canvas coordinates — canvas angles increase clockwise from the positive x-axis, so straight up is a quarter turn counter-clockwise from "facing right."

Drawing a rotated shape

Rather than computing each point of a rotated triangle by hand, the canvas has a transform stack: save() remembers where you were, translate()/rotate() move and spin the coordinate system itself, and restore() puts it back. So the ship's shape is drawn as if it always faces along the positive x-axis (angle 0) — the transform handles making it actually point the right way.

JavaScript
var COLOR_SHIP = '#e2e8f0';

function drawShip() {
  var cx = state.canvasCssW / 2, cy = state.canvasCssH / 2;
  var r = state.ship.radius;
  ctx.save();
  ctx.translate(cx, cy);
  ctx.rotate(state.ship.heading);

  ctx.beginPath();
  ctx.moveTo(r, 0);
  ctx.lineTo(-r * 0.7, r * 0.7);
  ctx.lineTo(-r * 0.35, 0);
  ctx.lineTo(-r * 0.7, -r * 0.7);
  ctx.closePath();
  ctx.strokeStyle = COLOR_SHIP;
  ctx.lineWidth = 2;
  ctx.stroke();
  ctx.restore();
}

The four points trace a simple arrowhead: a nose out at (r, 0), two back corners, and a notch pulled in at the tail so it reads as a ship rather than a plain triangle. Note it's drawn at the exact center of the canvas every time, regardless of any x/y the ship might have — lesson 4 explains why that's deliberate.

A little atmosphere

Two small functions add a starfield: generateStars() scatters a handful of dots (scaled to the canvas area, so it looks similar on any screen size) whenever the canvas resizes, and drawBackground() paints them each frame, before the ship.

JavaScript
var COLOR_STAR = 'rgba(148, 163, 184, 0.5)';
var stars = [];

function generateStars() {
  stars = [];
  var count = Math.round((state.canvasCssW * state.canvasCssH) / 4000);
  for (var i = 0; i < count; i++) {
    stars.push({
      x: Math.random() * state.canvasCssW,
      y: Math.random() * state.canvasCssH,
      r: Math.random() * 1.2 + 0.3
    });
  }
}

function drawBackground() {
  ctx.fillStyle = COLOR_STAR;
  for (var i = 0; i < stars.length; i++) {
    var s = stars[i];
    ctx.beginPath();
    ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
    ctx.fill();
  }
}

generateStars() gets called at the end of resizeCanvas() from lesson 1, and render() now calls drawBackground() then drawShip(), in that order, so the ship is always drawn on top.

<script>
(function () {
  "use strict";

  // ---- Constants ----------------------------------------------------
  var SHIP_RADIUS = 12;
  var COLOR_SHIP = '#e2e8f0';
  var COLOR_STAR = 'rgba(148, 163, 184, 0.5)';
  var MAX_DT = 0.05;

  // ---- DOM ------------------------------------------------------------
  var canvasWrap = document.getElementById('canvasWrap');
  var canvas = document.getElementById('gameCanvas');
  var ctx = canvas.getContext('2d');

  // ---- State ------------------------------------------------------------
  var state = {
    ship: {
      heading: -Math.PI / 2, // pointing "up"
      radius: SHIP_RADIUS
    },
    canvasCssW: 0, canvasCssH: 0, canvasDpr: 1,
    timing: { lastFrameTime: 0 }
  };
  window.__astDebugState = state;

  var stars = [];

  // ---- Canvas sizing ------------------------------------------------------
  function resizeCanvas() {
    var rect = canvasWrap.getBoundingClientRect();
    var dpr = window.devicePixelRatio || 1;
    var w = Math.round(rect.width);
    var h = Math.round(rect.height);
    if (w === state.canvasCssW && h === state.canvasCssH && dpr === state.canvasDpr) return;
    state.canvasCssW = w;
    state.canvasCssH = h;
    state.canvasDpr = dpr;
    canvas.width = Math.round(w * dpr);
    canvas.height = Math.round(h * dpr);
    canvas.style.width = w + 'px';
    canvas.style.height = h + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    generateStars();
  }

  function generateStars() {
    stars = [];
    var count = Math.round((state.canvasCssW * state.canvasCssH) / 4000);
    for (var i = 0; i < count; i++) {
      stars.push({
        x: Math.random() * state.canvasCssW,
        y: Math.random() * state.canvasCssH,
        r: Math.random() * 1.2 + 0.3
      });
    }
  }

  var resizeTimer = null;
  function onResize() {
    clearTimeout(resizeTimer);
    resizeTimer = setTimeout(resizeCanvas, 150);
  }

  // ---- Rendering ------------------------------------------------------
  function drawBackground() {
    ctx.fillStyle = COLOR_STAR;
    for (var i = 0; i < stars.length; i++) {
      var s = stars[i];
      ctx.beginPath();
      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  function drawShip() {
    var cx = state.canvasCssW / 2, cy = state.canvasCssH / 2;
    var r = state.ship.radius;
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate(state.ship.heading);

    ctx.beginPath();
    ctx.moveTo(r, 0);
    ctx.lineTo(-r * 0.7, r * 0.7);
    ctx.lineTo(-r * 0.35, 0);
    ctx.lineTo(-r * 0.7, -r * 0.7);
    ctx.closePath();
    ctx.strokeStyle = COLOR_SHIP;
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.restore();
  }

  function render() {
    ctx.clearRect(0, 0, state.canvasCssW, state.canvasCssH);
    drawBackground();
    drawShip();
  }

  // ---- Game loop ------------------------------------------------------
  var rafId = null;
  function frame(ts) {
    rafId = requestAnimationFrame(frame);
    if (!state.timing.lastFrameTime) { state.timing.lastFrameTime = ts; return; }
    var dt = Math.min((ts - state.timing.lastFrameTime) / 1000, MAX_DT);
    state.timing.lastFrameTime = ts;
    render();
  }

  window.addEventListener('resize', onResize);
  window.addEventListener('orientationchange', onResize);

  // ---- Init ------------------------------------------------------
  resizeCanvas();
  rafId = requestAnimationFrame(frame);
})();
</script>
