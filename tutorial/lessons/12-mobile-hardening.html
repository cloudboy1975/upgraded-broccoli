Mobile Hardening

The control bar from last lesson works, right up until it meets a real phone. These three fixes are applied here in the order they were actually discovered shipping this exact game: a button getting stuck held down, a magnifier popup on long-press, and accidental page pinch-zoom. Each one is a real bug with a specific, traceable cause — not defensive code added on spec.

Bug 1: a button gets stuck held down

bindHoldZone's pointerdown handler only ever released state on pointerup/pointercancel. If a touch gets interrupted without one of those firing — the OS backgrounds the tab mid-touch, or a long-press triggers the browser's own text-selection takeover — the zone's held flag stays stuck true forever, and worse, a fresh tap couldn't fix it: the old code bailed out immediately whenever activeId was already set, silently ignoring the very tap a player would make trying to recover.

The fix is two layers. First, a shared hardRelease() that every zone registers into heldZoneResets, called from visibilitychange and blur — so backgrounding the tab or losing window focus force-releases everything immediately. Second, a per-zone watchdog timeout as a last resort for any other missed-event case a browser might produce. And critically, pointerdown now calls hardRelease() unconditionally before starting a new press, instead of bailing out when a session was already active — so a stuck zone recovers the instant it's tapped again, with no need to wait for any of the other layers.

JavaScript
var HOLD_WATCHDOG_MS = 12000; // no legitimate hold needs to last this long
var heldZoneResets = [];

function bindHoldZone(el, setFlag) {
  var activeId = null;
  var watchdogTimer = null;

  function hardRelease() {
    if (activeId === null) return;
    activeId = null;
    clearTimeout(watchdogTimer);
    el.classList.remove('active');
    setFlag(false);
    composeInput();
  }

  el.addEventListener('pointerdown', function (e) {
    hardRelease(); // release any stale session instead of silently ignoring the new press
    activeId = e.pointerId;
    try { el.setPointerCapture(activeId); } catch (err) {}
    el.classList.add('active');
    setFlag(true);
    composeInput();
    clearTimeout(watchdogTimer);
    watchdogTimer = setTimeout(hardRelease, HOLD_WATCHDOG_MS);
  });
  function release(e) {
    if (e.pointerId !== activeId) return;
    hardRelease();
  }
  el.addEventListener('pointerup', release);
  el.addEventListener('pointercancel', release);
  el.addEventListener('lostpointercapture', release);

  heldZoneResets.push(hardRelease);
}

});

Bug 2: a magnifier pops up on long-press

iOS Safari's native long-press-to-select gesture runs as its own touch gesture recognizer, independent of Pointer Events entirely — the CSS hardening already on .zone (user-select, touch-callout) turned out not to be enough once the element also had active JS touch handling. The actual fix: a non-passive touchstart listener that calls preventDefault(), which is the specific call that tells WebKit to cancel its native gesture for that touch. pointerdown gets a matching preventDefault() as a second layer.

JavaScript
function suppressCallout(el) {
  el.addEventListener('touchstart', function (e) { e.preventDefault(); }, { passive: false });
}
// called once per zone: suppressCallout(zoneLeft); suppressCallout(zoneMid); suppressCallout(zoneRight);
// and el.addEventListener('pointerdown', function (e) { e.preventDefault(); ... }

Belt-and-suspenders: contextmenu and selectstart are also prevented on the whole control bar, for any mobile browser that still shows a callout menu despite the CSS and the touchstart fix above.

Bug 3: pinching the screen zooms the whole page

iOS Safari deliberately ignores maximum-scale/ user-scalable=no in the viewport meta tag — an accessibility policy Apple won't let a page override — so nothing in the page's <head> can fully block pinch-zoom. Two layers instead: cancel WebKit's native gesture events outright, and as a safety net, if a pinch gets through anyway, detect the zoomed-in state and snap it back to 1x shortly after, so a player can never get stuck zoomed in with no way back.

JavaScript
document.addEventListener('gesturestart', function (e) { e.preventDefault(); });
document.addEventListener('gesturechange', function (e) { e.preventDefault(); });
document.addEventListener('touchmove', function (e) {
  if (e.touches && e.touches.length > 1) e.preventDefault();
}, { passive: false });

(function () {
  var viewportMeta = document.querySelector('meta[name="viewport"]');
  var resetTimer = null;

  function forceResetZoom() {
    if (!viewportMeta) return;
    var original = viewportMeta.getAttribute('content');
    viewportMeta.setAttribute('content', original + ', shrink-to-fit=no');
    requestAnimationFrame(function () { viewportMeta.setAttribute('content', original); });
  }

  function scheduleZoomCheck() {
    clearTimeout(resetTimer);
    resetTimer = setTimeout(function () {
      if (window.visualViewport && window.visualViewport.scale > 1.02) forceResetZoom();
    }, 250);
  }

  if (window.visualViewport) {
    window.visualViewport.addEventListener('resize', scheduleZoomCheck);
  }
})();

      composeInput();

      e.preventDefault();

      // session (the previous touch never got a pointerup/pointercancel -
      // exactly the "stuck button" case) or a second finger on the same
      // zone. Either way, release whatever's recorded and start fresh
      // rather than silently ignoring the new press - that silent-ignore

      composeInput();

      if (e.pointerId !== activeId) return;
      hardRelease();

    el.addEventListener('pointercancel', release);
    el.addEventListener('lostpointercapture', release);

  });

  // Belt-and-suspenders against long-press text selection / callout menus
  // some mobile browsers still show on custom buttons despite the CSS above.
  controlBar.addEventListener('contextmenu', function (e) { e.preventDefault(); });
  controlBar.addEventListener('selectstart', function (e) { e.preventDefault(); });

  // ---- Prevent / auto-recover from accidental page pinch-zoom ------------------------------------------------------
  // iOS Safari ignores maximum-scale/user-scalable in the viewport meta
  // tag (deliberate, for accessibility - Apple won't let a page fully
  // block pinch-zoom that way) and touch-action:none isn't 100% reliable
  // against pinch specifically on iOS either. Two layers: stop the
  // WebKit gesture events outright, and as a safety net, snap the zoom
  // back to 1x shortly after any pinch that gets through anyway, so a
  // player is never stuck zoomed in with no way back.
  document.addEventListener('gesturestart', function (e) { e.preventDefault(); });
  document.addEventListener('gesturechange', function (e) { e.preventDefault(); });
  document.addEventListener('touchmove', function (e) {
    if (e.touches && e.touches.length > 1) e.preventDefault();
  }, { passive: false });

  (function () {
    var viewportMeta = document.querySelector('meta[name="viewport"]');
    var resetTimer = null;

    function forceResetZoom() {
      if (!viewportMeta) return;
      var original = viewportMeta.getAttribute('content');
      viewportMeta.setAttribute('content', original + ', shrink-to-fit=no');
      requestAnimationFrame(function () { viewportMeta.setAttribute('content', original); });
    }

    function scheduleZoomCheck() {
      clearTimeout(resetTimer);
      resetTimer = setTimeout(function () {
        if (window.visualViewport && window.visualViewport.scale > 1.02) forceResetZoom();
      }, 250);
    }

    if (window.visualViewport) {
      window.visualViewport.addEventListener('resize', scheduleZoomCheck);
    }
  })();

  // ---- Math helpers ------------------------------------------------------
  function randRange(range) { return range[0] + Math.random() * (range[1] - range[0]); }

  function visibleRadius() {
    return Math.sqrt(state.canvasCssW * state.canvasCssW + state.canvasCssH * state.canvasCssH) / 2;
  }
  function spawnRingRadius() { return visibleRadius() + 80; }
  function despawnRadius() { return visibleRadius() * 2.5; }
  function targetRockCount() { return Math.min(6 + Math.floor(state.difficulty.elapsed / 30), 14); }
  function speedMultiplier() { return 1 + Math.min(state.difficulty.elapsed / 60, 1) * 0.5; }

  function worldToScreen(wx, wy) {
    return {
      x: wx - state.ship.x + state.canvasCssW / 2,
      y: wy - state.ship.y + state.canvasCssH / 2
    };
  }

  // ---- Rocks ------------------------------------------------------
  function makeRockShape() {
    var points = [];
    var n = 10 + Math.floor(Math.random() * 4);
    for (var i = 0; i < n; i++) {
      points.push({
        angle: (i / n) * Math.PI * 2,
        r: 0.75 + Math.random() * 0.5
      });
    }
    return points;
  }

  function createRock(size, x, y, vx, vy) {
    return {
      id: state.nextRockId++,
      x: x, y: y, vx: vx, vy: vy,
      rotation: Math.random() * Math.PI * 2,
      rotationSpeed: (Math.random() - 0.5) * 1.6,
      size: size,
      radius: ROCK_SIZES[size].radius,
      shapePoints: makeRockShape()
    };
  }

  // ---- Rock spawn / despawn ------------------------------------------------------
  function seedRocks() {
    var minDist = visibleRadius() * 0.6;
    var maxDist = despawnRadius() * 0.5;
    for (var i = 0; i < 6; i++) {
      var angle = Math.random() * Math.PI * 2;
      var dist = minDist + Math.random() * (maxDist - minDist);
      var x = state.ship.x + Math.cos(angle) * dist;
      var y = state.ship.y + Math.sin(angle) * dist;
      var dirAngle = Math.random() * Math.PI * 2;
      var speed = randRange(ROCK_SIZES.large.speedRange) * speedMultiplier();
      state.rocks.push(createRock('large', x, y, Math.cos(dirAngle) * speed, Math.sin(dirAngle) * speed));
    }
  }

  function trySpawnRocks(dt) {
    state.spawn.timer -= dt;
    if (state.spawn.timer > 0) return;
    state.spawn.timer = SPAWN_CHECK_INTERVAL;
    if (state.rocks.length >= targetRockCount()) return;

    var angle = Math.random() * Math.PI * 2;
    var dist = spawnRingRadius();
    var x = state.ship.x + Math.cos(angle) * dist;
    var y = state.ship.y + Math.sin(angle) * dist;
    var inward = angle + Math.PI + (Math.random() - 0.5) * 1.0;
    var speed = randRange(ROCK_SIZES.large.speedRange) * speedMultiplier();
    state.rocks.push(createRock('large', x, y, Math.cos(inward) * speed, Math.sin(inward) * speed));
  }

  function despawnFarRocks() {
    var dr = despawnRadius();
    for (var i = state.rocks.length - 1; i >= 0; i--) {
      var rk = state.rocks[i];
      var dx = rk.x - state.ship.x, dy = rk.y - state.ship.y;
      if (dx * dx + dy * dy > dr * dr) state.rocks.splice(i, 1);
    }
  }

  // ---- Physics update ------------------------------------------------------
  function updateShip(dt) {
    var ship = state.ship;
    if (state.input.left) ship.heading -= TURN_RATE * dt;
    if (state.input.right) ship.heading += TURN_RATE * dt;
    ship.thrusting = state.input.thrust;
    if (ship.thrusting) {
      ship.vx += Math.cos(ship.heading) * THRUST_ACCEL * dt;
      ship.vy += Math.sin(ship.heading) * THRUST_ACCEL * dt;
    }
    var damp = Math.pow(1 - DAMPING_PER_SECOND, dt);
    ship.vx *= damp;
    ship.vy *= damp;
    var speed = Math.sqrt(ship.vx * ship.vx + ship.vy * ship.vy);
    if (speed > MAX_SPEED) {
      var scale = MAX_SPEED / speed;
      ship.vx *= scale;
      ship.vy *= scale;
    }
    ship.x += ship.vx * dt;
    ship.y += ship.vy * dt;
  }

  function updateRocks(dt) {
    for (var i = 0; i < state.rocks.length; i++) {
      var rk = state.rocks[i];
      rk.x += rk.vx * dt;
      rk.y += rk.vy * dt;
      rk.rotation += rk.rotationSpeed * dt;
    }
  }

  function updateBullets(dt) {
    for (var i = state.bullets.length - 1; i >= 0; i--) {
      var b = state.bullets[i];
      b.x += b.vx * dt;
      b.y += b.vy * dt;
      b.ttl -= dt;
      if (b.ttl <= 0) state.bullets.splice(i, 1);
    }
  }

  // ---- Collisions ------------------------------------------------------
  function handleCollisions() {
    for (var bi = state.bullets.length - 1; bi >= 0; bi--) {
      var b = state.bullets[bi];
      for (var ri = state.rocks.length - 1; ri >= 0; ri--) {
        var rk = state.rocks[ri];
        var dx = b.x - rk.x, dy = b.y - rk.y;
        var rr = rk.radius + BULLET_RADIUS;
        if (dx * dx + dy * dy <= rr * rr) {
          state.bullets.splice(bi, 1);
          splitRock(ri);
          break;
        }
      }
    }

    if (!isInvulnerable()) {
      for (var i = 0; i < state.rocks.length; i++) {
        var rock = state.rocks[i];
        var sx = state.ship.x - rock.x, sy = state.ship.y - rock.y;
        var sr = state.ship.radius + rock.radius;
        if (sx * sx + sy * sy <= sr * sr) {
          shipHit();
          break;
        }
      }
    }
  }

  function isInvulnerable() { return performance.now() < state.ship.invulnerableUntil; }

  function shipHit() {
    state.lives--;
    if (state.lives <= 0) {
      state.lives = 0;
      endGame();
    } else {
      respawnShip();
    }
    renderHud();
  }

  function respawnShip() {
    state.ship.vx = 0;
    state.ship.vy = 0;
    state.ship.heading = -Math.PI / 2;
    state.ship.invulnerableUntil = performance.now() + RESPAWN_INVULN_MS;
  }

  function endGame() {
    state.phase = 'gameover';
    overlayScoreEl.textContent = state.score;
    overlayBestEl.textContent = state.highScore;
    overlayEl.classList.add('visible');
  }

  function splitRock(index) {
    var rock = state.rocks[index];
    state.rocks.splice(index, 1);
    var info = ROCK_SIZES[rock.size];
    addScore(info.score);
    if (!info.child) return;
    var baseAngle = (rock.vx || rock.vy) ? Math.atan2(rock.vy, rock.vx) : Math.random() * Math.PI * 2;
    var childInfo = ROCK_SIZES[info.child];
    for (var k = 0; k < 2; k++) {
      var spread = 0.5 + Math.random() * 0.3;
      var ang = baseAngle + (k === 0 ? spread : -spread);
      var speed = randRange(childInfo.speedRange) * speedMultiplier();
      state.rocks.push(createRock(info.child, rock.x, rock.y, Math.cos(ang) * speed, Math.sin(ang) * speed));
    }
  }

  function addScore(n) {
    state.score += n;
    if (state.score > state.highScore) {
      state.highScore = state.score;
      saveHighScore();
    }
    renderHud();
  }

  // ---- Persistence ------------------------------------------------------
  function loadHighScore() {
    try {
      var raw = localStorage.getItem(HIGH_SCORE_KEY);
      var n = parseInt(raw, 10);
      state.highScore = (isFinite(n) && n > 0) ? n : 0;
    } catch (e) { /* ignore storage errors (private browsing, quota, etc.) */ }
  }

  function saveHighScore() {
    try { localStorage.setItem(HIGH_SCORE_KEY, String(state.highScore)); } catch (e) {}
  }

  function resetBest() {
    state.highScore = 0;
    saveHighScore();
    renderHud();
  }

  // ---- HUD ------------------------------------------------------
  function renderHud() {
    scoreValEl.textContent = state.score;
    livesValEl.textContent = state.lives;
    bestValEl.textContent = state.highScore;
    phaseDotEl.className = 'turn-dot ' + (state.phase === 'playing' ? 'playing' : 'gameover');
    statusLabelEl.textContent = state.phase === 'playing' ? 'Playing' : 'Game Over';
  }

  // ---- Reset / restart ------------------------------------------------------
  function resetGame() {
    state.phase = 'playing';
    state.ship.x = 0; state.ship.y = 0;
    state.ship.vx = 0; state.ship.vy = 0;
    state.ship.heading = -Math.PI / 2;
    state.ship.thrusting = false;
    state.ship.invulnerableUntil = performance.now() + RESPAWN_INVULN_MS;
    state.rocks = [];
    state.bullets = [];
    state.lives = 3;
    state.score = 0;
    state.spawn.timer = 0;
    state.difficulty.elapsed = 0;
    state.nextRockId = 1;
    state.timing.lastFrameTime = 0;
    seedRocks();
    overlayEl.classList.remove('visible');
    renderHud();
  }

  // ---- Rendering ------------------------------------------------------
  function drawBackground() {
    ctx.fillStyle = COLOR_STAR;
    for (var i = 0; i < stars.length; i++) {
      var s = stars[i];
      ctx.beginPath();
      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  function drawRock(rock) {
    var p = worldToScreen(rock.x, rock.y);
    ctx.save();
    ctx.translate(p.x, p.y);
    ctx.rotate(rock.rotation);
    ctx.beginPath();
    for (var i = 0; i < rock.shapePoints.length; i++) {
      var pt = rock.shapePoints[i];
      var rad = rock.radius * pt.r;
      var px = Math.cos(pt.angle) * rad;
      var py = Math.sin(pt.angle) * rad;
      if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
    }
    ctx.closePath();
    ctx.fillStyle = COLOR_ROCK_FILL;
    ctx.fill();
    ctx.strokeStyle = COLOR_ROCK_STROKE;
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.restore();
  }

  function drawBullet(b) {
    var p = worldToScreen(b.x, b.y);
    ctx.beginPath();
    ctx.arc(p.x, p.y, BULLET_RADIUS, 0, Math.PI * 2);
    ctx.fillStyle = COLOR_BULLET;
    ctx.fill();
  }

  function drawShip() {
    if (isInvulnerable() && Math.floor(performance.now() / 100) % 2 === 0) return;
    var cx = state.canvasCssW / 2, cy = state.canvasCssH / 2;
    var r = state.ship.radius;
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate(state.ship.heading);

    if (state.ship.thrusting) {
      var flameLen = r * (0.8 + Math.random() * 0.5);
      ctx.beginPath();
      ctx.moveTo(-r * 0.35, r * 0.4);
      ctx.lineTo(-r * 0.35 - flameLen, 0);
      ctx.lineTo(-r * 0.35, -r * 0.4);
      ctx.closePath();
      ctx.fillStyle = COLOR_FLAME;
      ctx.fill();
    }

    ctx.beginPath();
    ctx.moveTo(r, 0);
    ctx.lineTo(-r * 0.7, r * 0.7);
    ctx.lineTo(-r * 0.35, 0);
    ctx.lineTo(-r * 0.7, -r * 0.7);
    ctx.closePath();
    ctx.strokeStyle = COLOR_SHIP;
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.restore();
  }

  function render() {
    ctx.clearRect(0, 0, state.canvasCssW, state.canvasCssH);
    drawBackground();
    for (var i = 0; i < state.rocks.length; i++) drawRock(state.rocks[i]);
    for (var j = 0; j < state.bullets.length; j++) drawBullet(state.bullets[j]);
    drawShip();
  }

  // ---- Game loop ------------------------------------------------------
  function update(dt) {
    updateShip(dt);
    updateRocks(dt);
    updateBullets(dt);
    trySpawnRocks(dt);
    despawnFarRocks();
    handleCollisions();
    state.difficulty.elapsed += dt;
  }

  var rafId = null;
  function frame(ts) {
    rafId = requestAnimationFrame(frame);
    if (!state.timing.lastFrameTime) { state.timing.lastFrameTime = ts; return; }
    var dt = Math.min((ts - state.timing.lastFrameTime) / 1000, MAX_DT);
    state.timing.lastFrameTime = ts;
    if (state.phase === 'playing') update(dt);
    render();
  }

  window.addEventListener('resize', onResize);
  window.addEventListener('orientationchange', onResize);
  restartBtn.addEventListener('click', resetGame);
  overlayRestartBtn.addEventListener('click', resetGame);
  resetBestBtn.addEventListener('click', resetBest);

  // ---- Init ------------------------------------------------------
  resizeCanvas();
  loadHighScore();
  resetGame(); // resetGame() already seeds the rock field, so it replaces the old seedRocks() call here
  rafId = requestAnimationFrame(frame);
})();
</script>
