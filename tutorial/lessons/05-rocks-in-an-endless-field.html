Rocks in an Endless Field

This game doesn't wrap rocks around screen edges the way the original arcade cabinet did — there are no edges, since the world is unbounded and the camera just follows the ship (lesson 4). Instead, rocks spawn in a ring just outside what's visible and get cleaned up once they're far enough behind you that they'd never be seen again. This lesson replaces last lesson's single parked rock with that real system.

Seeding the field

When the game starts, seedRocks() scatters six large rocks at random angles around the ship, each already moving in a random direction. Scattering them between 60% of the visible radius and half the despawn radius keeps the very first moment from either dropping a rock right on top of you or spawning one so far out it's invisible.

JavaScript
function randRange(range) { return range[0] + Math.random() * (range[1] - range[0]); }

function visibleRadius() {
  return Math.sqrt(state.canvasCssW * state.canvasCssW + state.canvasCssH * state.canvasCssH) / 2;
}
function despawnRadius() { return visibleRadius() * 2.5; }

function seedRocks() {
  var minDist = visibleRadius() * 0.6;
  var maxDist = despawnRadius() * 0.5;
  for (var i = 0; i < 6; i++) {
    var angle = Math.random() * Math.PI * 2;
    var dist = minDist + Math.random() * (maxDist - minDist);
    var x = state.ship.x + Math.cos(angle) * dist;
    var y = state.ship.y + Math.sin(angle) * dist;
    var dirAngle = Math.random() * Math.PI * 2;
    var speed = randRange(ROCK_SIZES.large.speedRange) * speedMultiplier();
    state.rocks.push(createRock('large', x, y, Math.cos(dirAngle) * speed, Math.sin(dirAngle) * speed));
  }
}

visibleRadius() is the radius of a circle just big enough to cover the whole canvas from its center — the Pythagorean diagonal, halved. Everything about spawning and despawning is expressed relative to it, so it automatically adapts to whatever size screen it's running on.

Spawning ahead, forever

Every half second (SPAWN_CHECK_INTERVAL), if the rock count is under a target that slowly ramps up over time, one more large rock spawns in a ring just past the edge of what's visible (spawnRingRadius()), aimed roughly — not exactly, there's some randomness in the angle — back toward the ship. That inward aim is what makes new rocks feel like a real threat instead of drifting past uselessly off in some random direction.

JavaScript
var SPAWN_CHECK_INTERVAL = 0.5; // seconds

function spawnRingRadius() { return visibleRadius() + 80; }
function targetRockCount() { return Math.min(6 + Math.floor(state.difficulty.elapsed / 30), 14); }
function speedMultiplier() { return 1 + Math.min(state.difficulty.elapsed / 60, 1) * 0.5; }

function trySpawnRocks(dt) {
  state.spawn.timer -= dt;
  if (state.spawn.timer > 0) return;
  state.spawn.timer = SPAWN_CHECK_INTERVAL;
  if (state.rocks.length >= targetRockCount()) return;

  var angle = Math.random() * Math.PI * 2;
  var dist = spawnRingRadius();
  var x = state.ship.x + Math.cos(angle) * dist;
  var y = state.ship.y + Math.sin(angle) * dist;
  var inward = angle + Math.PI + (Math.random() - 0.5) * 1.0;
  var speed = randRange(ROCK_SIZES.large.speedRange) * speedMultiplier();
  state.rocks.push(createRock('large', x, y, Math.cos(inward) * speed, Math.sin(inward) * speed));
}

targetRockCount() and speedMultiplier() both read state.difficulty.elapsed — total seconds played — so the field slowly gets denser and faster the longer you survive, capping out rather than spiraling out of control forever.

Cleaning up behind you

Without some cleanup, a rock spawned five minutes ago and drifting the opposite direction from the ship would still be sitting in the array, updated every frame, forever. despawnFarRocks() just removes anything that's drifted out past despawnRadius() — far enough that it could never be on screen again regardless of where the ship goes next.

JavaScript
function despawnFarRocks() {
  var dr = despawnRadius();
  for (var i = state.rocks.length - 1; i >= 0; i--) {
    var rk = state.rocks[i];
    var dx = rk.x - state.ship.x, dy = rk.y - state.ship.y;
    if (dx * dx + dy * dy > dr * dr) state.rocks.splice(i, 1);
  }
}

function updateRocks(dt) {
  for (var i = 0; i < state.rocks.length; i++) {
    var rk = state.rocks[i];
    rk.x += rk.vx * dt;
    rk.y += rk.vy * dt;
    rk.rotation += rk.rotationSpeed * dt;
  }
}

The loop counts backward specifically so splice() removing an element mid-loop doesn't skip the next one — a classic array-mutation-during-iteration bug avoided by iterating in reverse. update(dt) now calls updateRocks, trySpawnRocks, and despawnFarRocks every frame, alongside updateShip from lesson 3.

<script>
(function () {
  "use strict";

  // ---- Constants ----------------------------------------------------
  var SHIP_RADIUS = 12;
  var THRUST_ACCEL = 220;
  var MAX_SPEED = 260;
  var DAMPING_PER_SECOND = 0.5;
  var TURN_RATE = 3.2; // radians/sec
  var MAX_DT = 0.05;
  var SPAWN_CHECK_INTERVAL = 0.5; // seconds

  var ROCK_SIZES = {
    large:  { radius: 46, speedRange: [30, 70],   score: 20,  child: 'medium' },
    medium: { radius: 27, speedRange: [50, 100],  score: 50,  child: 'small' },
    small:  { radius: 15, speedRange: [80, 140],  score: 100, child: null }
  };

  var COLOR_SHIP = '#e2e8f0';
  var COLOR_FLAME = '#f97316';
  var COLOR_STAR = 'rgba(148, 163, 184, 0.5)';
  var COLOR_ROCK_STROKE = '#334155';
  var COLOR_ROCK_FILL = 'rgba(148, 163, 184, 0.18)';

  // ---- DOM ------------------------------------------------------------
  var canvasWrap = document.getElementById('canvasWrap');
  var canvas = document.getElementById('gameCanvas');
  var ctx = canvas.getContext('2d');

  // ---- State ------------------------------------------------------------
  var state = {
    ship: {
      x: 0, y: 0, vx: 0, vy: 0,
      heading: -Math.PI / 2,
      thrusting: false,
      radius: SHIP_RADIUS
    },
    rocks: [],
    nextRockId: 1,
    input: { left: false, right: false, thrust: false },
    spawn: { timer: 0 },
    difficulty: { elapsed: 0 },
    canvasCssW: 0, canvasCssH: 0, canvasDpr: 1,
    timing: { lastFrameTime: 0 }
  };
  window.__astDebugState = state;

  var stars = [];

  // ---- Canvas sizing ------------------------------------------------------
  function resizeCanvas() {
    var rect = canvasWrap.getBoundingClientRect();
    var dpr = window.devicePixelRatio || 1;
    var w = Math.round(rect.width);
    var h = Math.round(rect.height);
    if (w === state.canvasCssW && h === state.canvasCssH && dpr === state.canvasDpr) return;
    state.canvasCssW = w;
    state.canvasCssH = h;
    state.canvasDpr = dpr;
    canvas.width = Math.round(w * dpr);
    canvas.height = Math.round(h * dpr);
    canvas.style.width = w + 'px';
    canvas.style.height = h + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    generateStars();
  }

  function generateStars() {
    stars = [];
    var count = Math.round((state.canvasCssW * state.canvasCssH) / 4000);
    for (var i = 0; i < count; i++) {
      stars.push({
        x: Math.random() * state.canvasCssW,
        y: Math.random() * state.canvasCssH,
        r: Math.random() * 1.2 + 0.3
      });
    }
  }

  var resizeTimer = null;
  function onResize() {
    clearTimeout(resizeTimer);
    resizeTimer = setTimeout(resizeCanvas, 150);
  }

  // ---- Input: keyboard ------------------------------------------------------
  function isLeftKey(code) { return code === 'ArrowLeft' || code === 'KeyA'; }
  function isRightKey(code) { return code === 'ArrowRight' || code === 'KeyD'; }
  function isThrustKey(code) { return code === 'Space' || code === 'ArrowUp' || code === 'KeyW'; }

  document.addEventListener('keydown', function (e) {
    if (isThrustKey(e.code)) e.preventDefault();
    if (isLeftKey(e.code)) state.input.left = true;
    else if (isRightKey(e.code)) state.input.right = true;
    else if (isThrustKey(e.code)) state.input.thrust = true;
  });

  document.addEventListener('keyup', function (e) {
    if (isLeftKey(e.code)) state.input.left = false;
    else if (isRightKey(e.code)) state.input.right = false;
    else if (isThrustKey(e.code)) state.input.thrust = false;
  });

  // ---- Math helpers ------------------------------------------------------
  function randRange(range) { return range[0] + Math.random() * (range[1] - range[0]); }

  function visibleRadius() {
    return Math.sqrt(state.canvasCssW * state.canvasCssW + state.canvasCssH * state.canvasCssH) / 2;
  }
  function spawnRingRadius() { return visibleRadius() + 80; }
  function despawnRadius() { return visibleRadius() * 2.5; }
  function targetRockCount() { return Math.min(6 + Math.floor(state.difficulty.elapsed / 30), 14); }
  function speedMultiplier() { return 1 + Math.min(state.difficulty.elapsed / 60, 1) * 0.5; }

  function worldToScreen(wx, wy) {
    return {
      x: wx - state.ship.x + state.canvasCssW / 2,
      y: wy - state.ship.y + state.canvasCssH / 2
    };
  }

  // ---- Rocks ------------------------------------------------------
  function makeRockShape() {
    var points = [];
    var n = 10 + Math.floor(Math.random() * 4);
    for (var i = 0; i < n; i++) {
      points.push({
        angle: (i / n) * Math.PI * 2,
        r: 0.75 + Math.random() * 0.5
      });
    }
    return points;
  }

  function createRock(size, x, y, vx, vy) {
    return {
      id: state.nextRockId++,
      x: x, y: y, vx: vx, vy: vy,
      rotation: Math.random() * Math.PI * 2,
      rotationSpeed: (Math.random() - 0.5) * 1.6,
      size: size,
      radius: ROCK_SIZES[size].radius,
      shapePoints: makeRockShape()
    };
  }

  // ---- Rock spawn / despawn ------------------------------------------------------
  function seedRocks() {
    var minDist = visibleRadius() * 0.6;
    var maxDist = despawnRadius() * 0.5;
    for (var i = 0; i < 6; i++) {
      var angle = Math.random() * Math.PI * 2;
      var dist = minDist + Math.random() * (maxDist - minDist);
      var x = state.ship.x + Math.cos(angle) * dist;
      var y = state.ship.y + Math.sin(angle) * dist;
      var dirAngle = Math.random() * Math.PI * 2;
      var speed = randRange(ROCK_SIZES.large.speedRange) * speedMultiplier();
      state.rocks.push(createRock('large', x, y, Math.cos(dirAngle) * speed, Math.sin(dirAngle) * speed));
    }
  }

  function trySpawnRocks(dt) {
    state.spawn.timer -= dt;
    if (state.spawn.timer > 0) return;
    state.spawn.timer = SPAWN_CHECK_INTERVAL;
    if (state.rocks.length >= targetRockCount()) return;

    var angle = Math.random() * Math.PI * 2;
    var dist = spawnRingRadius();
    var x = state.ship.x + Math.cos(angle) * dist;
    var y = state.ship.y + Math.sin(angle) * dist;
    var inward = angle + Math.PI + (Math.random() - 0.5) * 1.0;
    var speed = randRange(ROCK_SIZES.large.speedRange) * speedMultiplier();
    state.rocks.push(createRock('large', x, y, Math.cos(inward) * speed, Math.sin(inward) * speed));
  }

  function despawnFarRocks() {
    var dr = despawnRadius();
    for (var i = state.rocks.length - 1; i >= 0; i--) {
      var rk = state.rocks[i];
      var dx = rk.x - state.ship.x, dy = rk.y - state.ship.y;
      if (dx * dx + dy * dy > dr * dr) state.rocks.splice(i, 1);
    }
  }

  // ---- Physics update ------------------------------------------------------
  function updateShip(dt) {
    var ship = state.ship;
    if (state.input.left) ship.heading -= TURN_RATE * dt;
    if (state.input.right) ship.heading += TURN_RATE * dt;
    ship.thrusting = state.input.thrust;
    if (ship.thrusting) {
      ship.vx += Math.cos(ship.heading) * THRUST_ACCEL * dt;
      ship.vy += Math.sin(ship.heading) * THRUST_ACCEL * dt;
    }
    var damp = Math.pow(1 - DAMPING_PER_SECOND, dt);
    ship.vx *= damp;
    ship.vy *= damp;
    var speed = Math.sqrt(ship.vx * ship.vx + ship.vy * ship.vy);
    if (speed > MAX_SPEED) {
      var scale = MAX_SPEED / speed;
      ship.vx *= scale;
      ship.vy *= scale;
    }
    ship.x += ship.vx * dt;
    ship.y += ship.vy * dt;
  }

  function updateRocks(dt) {
    for (var i = 0; i < state.rocks.length; i++) {
      var rk = state.rocks[i];
      rk.x += rk.vx * dt;
      rk.y += rk.vy * dt;
      rk.rotation += rk.rotationSpeed * dt;
    }
  }

  // ---- Rendering ------------------------------------------------------
  function drawBackground() {
    ctx.fillStyle = COLOR_STAR;
    for (var i = 0; i < stars.length; i++) {
      var s = stars[i];
      ctx.beginPath();
      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  function drawRock(rock) {
    var p = worldToScreen(rock.x, rock.y);
    ctx.save();
    ctx.translate(p.x, p.y);
    ctx.rotate(rock.rotation);
    ctx.beginPath();
    for (var i = 0; i < rock.shapePoints.length; i++) {
      var pt = rock.shapePoints[i];
      var rad = rock.radius * pt.r;
      var px = Math.cos(pt.angle) * rad;
      var py = Math.sin(pt.angle) * rad;
      if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
    }
    ctx.closePath();
    ctx.fillStyle = COLOR_ROCK_FILL;
    ctx.fill();
    ctx.strokeStyle = COLOR_ROCK_STROKE;
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.restore();
  }

  function drawShip() {
    var cx = state.canvasCssW / 2, cy = state.canvasCssH / 2;
    var r = state.ship.radius;
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate(state.ship.heading);

    if (state.ship.thrusting) {
      var flameLen = r * (0.8 + Math.random() * 0.5);
      ctx.beginPath();
      ctx.moveTo(-r * 0.35, r * 0.4);
      ctx.lineTo(-r * 0.35 - flameLen, 0);
      ctx.lineTo(-r * 0.35, -r * 0.4);
      ctx.closePath();
      ctx.fillStyle = COLOR_FLAME;
      ctx.fill();
    }

    ctx.beginPath();
    ctx.moveTo(r, 0);
    ctx.lineTo(-r * 0.7, r * 0.7);
    ctx.lineTo(-r * 0.35, 0);
    ctx.lineTo(-r * 0.7, -r * 0.7);
    ctx.closePath();
    ctx.strokeStyle = COLOR_SHIP;
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.restore();
  }

  function render() {
    ctx.clearRect(0, 0, state.canvasCssW, state.canvasCssH);
    drawBackground();
    for (var i = 0; i < state.rocks.length; i++) drawRock(state.rocks[i]);
    drawShip();
  }

  // ---- Game loop ------------------------------------------------------
  function update(dt) {
    updateShip(dt);
    updateRocks(dt);
    trySpawnRocks(dt);
    despawnFarRocks();
    state.difficulty.elapsed += dt;
  }

  var rafId = null;
  function frame(ts) {
    rafId = requestAnimationFrame(frame);
    if (!state.timing.lastFrameTime) { state.timing.lastFrameTime = ts; return; }
    var dt = Math.min((ts - state.timing.lastFrameTime) / 1000, MAX_DT);
    state.timing.lastFrameTime = ts;
    update(dt);
    render();
  }

  window.addEventListener('resize', onResize);
  window.addEventListener('orientationchange', onResize);

  // ---- Init ------------------------------------------------------
  resizeCanvas();
  seedRocks();
  rafId = requestAnimationFrame(frame);
})();
</script>
