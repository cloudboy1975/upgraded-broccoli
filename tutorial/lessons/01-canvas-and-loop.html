The Canvas and the Game Loop

Every version of Asteroids in this collection starts the same way: a <canvas> sized to fill its box, and a loop that keeps redrawing it at the screen's refresh rate. Nothing moves yet — this lesson is just about getting that skeleton running correctly, including on a phone, where the canvas has to handle high-DPI screens and window resizes without turning blurry.

A canvas that fills its box

The canvas lives inside a wrapper div rather than being sized directly, so its CSS box (rounded corners, a fixed aspect ratio, a max width on large screens) is separate from the canvas's own pixel buffer, which the JavaScript below manages.

HTML + CSS
<div class="canvas-wrap" id="canvasWrap">
  <canvas id="gameCanvas"></canvas>
</div>

<style>

Sizing it for real (high-DPI displays)

A canvas actually has two sizes: the CSS size (how big it appears on screen) and the internal pixel buffer size (how many actual pixels you get to draw into). On a phone with devicePixelRatio of 2 or 3, using the CSS size for both leaves you drawing into a buffer with far fewer pixels than the screen has, and everything comes out blurry. resizeCanvas() measures the wrapper, multiplies by dpr for the actual buffer, and then uses ctx.setTransform so every draw call after this can keep working in plain CSS-pixel coordinates without thinking about DPI again.

JavaScript
function resizeCanvas() {
  var rect = canvasWrap.getBoundingClientRect();
  var dpr = window.devicePixelRatio || 1;
  var w = Math.round(rect.width);
  var h = Math.round(rect.height);
  if (w === state.canvasCssW && h === state.canvasCssH && dpr === state.canvasDpr) return;
  state.canvasCssW = w;
  state.canvasCssH = h;
  state.canvasDpr = dpr;
  canvas.width = Math.round(w * dpr);
  canvas.height = Math.round(h * dpr);
  canvas.style.width = w + 'px';
  canvas.style.height = h + 'px';
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
}

var resizeTimer = null;
function onResize() {
  clearTimeout(resizeTimer);
  resizeTimer = setTimeout(resizeCanvas, 150);
}

The early-return guard skips the (moderately expensive) resize work when nothing actually changed, and onResize debounces rapid-fire resize events — both matter more than they look, since mobile browsers fire resize events liberally while the on-screen keyboard shows/hides or the address bar collapses on scroll.

The game loop

requestAnimationFrame asks the browser to call your function right before the next repaint — but it doesn't promise a fixed interval. A phone under load, a background tab, or a 120Hz vs 60Hz screen all change how much time actually passed between frames. So instead of assuming "16ms per frame," the loop measures the real elapsed time (dt, in seconds) and every later lesson's physics multiplies by dt rather than a fixed constant — that's what makes the game run at the same real-world speed regardless of frame rate.

JavaScript
var MAX_DT = 0.05; // clamp dt so a stall (tab backgrounded, etc.) can't cause a huge jump

var rafId = null;
function frame(ts) {
  rafId = requestAnimationFrame(frame);
  if (!state.timing.lastFrameTime) { state.timing.lastFrameTime = ts; return; }
  var dt = Math.min((ts - state.timing.lastFrameTime) / 1000, MAX_DT);
  state.timing.lastFrameTime = ts;
  render();
}

The first call has no previous timestamp to diff against, so it just records one and returns — every frame after that computes a real dt. render() for now just clears the canvas; there's nothing to draw until the next lesson.

<script>
(function () {
  "use strict";

  var canvasWrap = document.getElementById('canvasWrap');
  var canvas = document.getElementById('gameCanvas');
  var ctx = canvas.getContext('2d');

  var state = {
    canvasCssW: 0, canvasCssH: 0, canvasDpr: 1,
    timing: { lastFrameTime: 0 }
  };
  window.__astDebugState = state; // read-only hook used by the tutorial's own tests

  var MAX_DT = 0.05;

  // ---- Canvas sizing ------------------------------------------------------
  function resizeCanvas() {
    var rect = canvasWrap.getBoundingClientRect();
    var dpr = window.devicePixelRatio || 1;
    var w = Math.round(rect.width);
    var h = Math.round(rect.height);
    if (w === state.canvasCssW && h === state.canvasCssH && dpr === state.canvasDpr) return;
    state.canvasCssW = w;
    state.canvasCssH = h;
    state.canvasDpr = dpr;
    canvas.width = Math.round(w * dpr);
    canvas.height = Math.round(h * dpr);
    canvas.style.width = w + 'px';
    canvas.style.height = h + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  }

  var resizeTimer = null;
  function onResize() {
    clearTimeout(resizeTimer);
    resizeTimer = setTimeout(resizeCanvas, 150);
  }

  // ---- Rendering ------------------------------------------------------
  function render() {
    ctx.clearRect(0, 0, state.canvasCssW, state.canvasCssH);
  }

  // ---- Game loop ------------------------------------------------------
  var rafId = null;
  function frame(ts) {
    rafId = requestAnimationFrame(frame);
    if (!state.timing.lastFrameTime) { state.timing.lastFrameTime = ts; return; }
    var dt = Math.min((ts - state.timing.lastFrameTime) / 1000, MAX_DT);
    state.timing.lastFrameTime = ts;
    render();
  }

  window.addEventListener('resize', onResize);
  window.addEventListener('orientationchange', onResize);

  // ---- Init ------------------------------------------------------
  resizeCanvas();
  rafId = requestAnimationFrame(frame);
})();
</script>
