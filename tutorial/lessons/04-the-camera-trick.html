The Camera Trick

Every rock, bullet, and eventually alien in this game will live in an unbounded "world" coordinate space — the ship's own x/y just keeps growing as it flies around. But lesson 2's drawShip() always draws the ship pinned to the exact center of the canvas. This lesson is the one function that reconciles those two facts — the trick this whole game is built around.

Two coordinate spaces

Think of it as a camera that always points at the ship. The ship has a real position in the world, but on screen it never moves — instead, everything else gets drawn offset by how far the ship has traveled. That offset is the whole function: worldToScreen() takes a world position and returns where it belongs on screen, given where the ship currently is.

JavaScript
function worldToScreen(wx, wy) {
  return {
    x: wx - state.ship.x + state.canvasCssW / 2,
    y: wy - state.ship.y + state.canvasCssH / 2
  };
}

When the ship is at the world origin, that's just "world position plus half the canvas" — the identity you'd expect. As state.ship.x/y grow while the ship flies around, every other object's screen position shifts by exactly the same amount in the opposite direction, which is what makes it look like the world is drifting past a stationary ship rather than the ship moving through a stationary world. Nothing about drawShip() needs to change at all — it was already hard-coded to draw at the canvas center back in lesson 2.

Proving it with an actual rock

Rather than fake this with a placeholder, this checkpoint uses the real asteroid code the next lesson builds on — createRock() makes a rock with an id, an irregular polygon shape, and a size; drawRock() draws it through worldToScreen(), exactly like every other world object will be drawn from here on.

JavaScript
var ROCK_SIZES = {
  large:  { radius: 46, speedRange: [30, 70],   score: 20,  child: 'medium' },
  medium: { radius: 27, speedRange: [50, 100],  score: 50,  child: 'small' },
  small:  { radius: 15, speedRange: [80, 140],  score: 100, child: null }
};

function makeRockShape() {
  var points = [];
  var n = 10 + Math.floor(Math.random() * 4);
  for (var i = 0; i < n; i++) {
    points.push({ angle: (i / n) * Math.PI * 2, r: 0.75 + Math.random() * 0.5 });
  }
  return points; // an irregular polygon, not a plain circle
}

function createRock(size, x, y, vx, vy) {
  return {
    id: state.nextRockId++,
    x: x, y: y, vx: vx, vy: vy,
    rotation: Math.random() * Math.PI * 2,
    rotationSpeed: (Math.random() - 0.5) * 1.6,
    size: size,
    radius: ROCK_SIZES[size].radius,
    shapePoints: makeRockShape()
  };
}
JavaScript
function drawRock(rock) {
  var p = worldToScreen(rock.x, rock.y); // <- the only line that cares about the camera
  ctx.save();
  ctx.translate(p.x, p.y);
  ctx.rotate(rock.rotation);
  ctx.beginPath();
  for (var i = 0; i < rock.shapePoints.length; i++) {
    var pt = rock.shapePoints[i];
    var rad = rock.radius * pt.r;
    var px = Math.cos(pt.angle) * rad;
    var py = Math.sin(pt.angle) * rad;
    if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
  }
  ctx.closePath();
  ctx.fillStyle = COLOR_ROCK_FILL;
  ctx.fill();
  ctx.strokeStyle = COLOR_ROCK_STROKE;
  ctx.lineWidth = 2;
  ctx.stroke();
  ctx.restore();
}

This checkpoint parks exactly one rock, with zero velocity, 130 units to the right of where the ship starts — close enough to peek into view immediately on most screens, but still requiring a turn and a thrust to actually reach: state.rocks.push(createRock('large', 130, 0, 0, 0)). That single line is scaffolding — the next lesson replaces it with a real spawning system that keeps a whole field of rocks drifting around you. Everything else on this page (worldToScreen, createRock, drawRock, ROCK_SIZES) is exactly what ships in the final game, unchanged.

Try it

Turn to face the rock and thrust toward it. The rock's world position never changes — watch it drift across the screen anyway as the ship's world position moves underneath it.

<script>
(function () {
  "use strict";

  // ---- Constants ----------------------------------------------------
  var SHIP_RADIUS = 12;
  var THRUST_ACCEL = 220;
  var MAX_SPEED = 260;
  var DAMPING_PER_SECOND = 0.5;
  var TURN_RATE = 3.2; // radians/sec
  var MAX_DT = 0.05;

  var ROCK_SIZES = {
    large:  { radius: 46, speedRange: [30, 70],   score: 20,  child: 'medium' },
    medium: { radius: 27, speedRange: [50, 100],  score: 50,  child: 'small' },
    small:  { radius: 15, speedRange: [80, 140],  score: 100, child: null }
  };

  var COLOR_SHIP = '#e2e8f0';
  var COLOR_FLAME = '#f97316';
  var COLOR_STAR = 'rgba(148, 163, 184, 0.5)';
  var COLOR_ROCK_STROKE = '#334155';
  var COLOR_ROCK_FILL = 'rgba(148, 163, 184, 0.18)';

  // ---- DOM ------------------------------------------------------------
  var canvasWrap = document.getElementById('canvasWrap');
  var canvas = document.getElementById('gameCanvas');
  var ctx = canvas.getContext('2d');

  // ---- State ------------------------------------------------------------
  var state = {
    ship: {
      x: 0, y: 0, vx: 0, vy: 0,
      heading: -Math.PI / 2,
      thrusting: false,
      radius: SHIP_RADIUS
    },
    rocks: [],
    nextRockId: 1,
    input: { left: false, right: false, thrust: false },
    canvasCssW: 0, canvasCssH: 0, canvasDpr: 1,
    timing: { lastFrameTime: 0 }
  };
  window.__astDebugState = state;

  var stars = [];

  // ---- Canvas sizing ------------------------------------------------------
  function resizeCanvas() {
    var rect = canvasWrap.getBoundingClientRect();
    var dpr = window.devicePixelRatio || 1;
    var w = Math.round(rect.width);
    var h = Math.round(rect.height);
    if (w === state.canvasCssW && h === state.canvasCssH && dpr === state.canvasDpr) return;
    state.canvasCssW = w;
    state.canvasCssH = h;
    state.canvasDpr = dpr;
    canvas.width = Math.round(w * dpr);
    canvas.height = Math.round(h * dpr);
    canvas.style.width = w + 'px';
    canvas.style.height = h + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    generateStars();
  }

  function generateStars() {
    stars = [];
    var count = Math.round((state.canvasCssW * state.canvasCssH) / 4000);
    for (var i = 0; i < count; i++) {
      stars.push({
        x: Math.random() * state.canvasCssW,
        y: Math.random() * state.canvasCssH,
        r: Math.random() * 1.2 + 0.3
      });
    }
  }

  var resizeTimer = null;
  function onResize() {
    clearTimeout(resizeTimer);
    resizeTimer = setTimeout(resizeCanvas, 150);
  }

  // ---- Input: keyboard ------------------------------------------------------
  function isLeftKey(code) { return code === 'ArrowLeft' || code === 'KeyA'; }
  function isRightKey(code) { return code === 'ArrowRight' || code === 'KeyD'; }
  function isThrustKey(code) { return code === 'Space' || code === 'ArrowUp' || code === 'KeyW'; }

  document.addEventListener('keydown', function (e) {
    if (isThrustKey(e.code)) e.preventDefault();
    if (isLeftKey(e.code)) state.input.left = true;
    else if (isRightKey(e.code)) state.input.right = true;
    else if (isThrustKey(e.code)) state.input.thrust = true;
  });

  document.addEventListener('keyup', function (e) {
    if (isLeftKey(e.code)) state.input.left = false;
    else if (isRightKey(e.code)) state.input.right = false;
    else if (isThrustKey(e.code)) state.input.thrust = false;
  });

  // ---- World <-> screen ------------------------------------------------------
  function worldToScreen(wx, wy) {
    return {
      x: wx - state.ship.x + state.canvasCssW / 2,
      y: wy - state.ship.y + state.canvasCssH / 2
    };
  }

  // ---- Rocks ------------------------------------------------------
  function makeRockShape() {
    var points = [];
    var n = 10 + Math.floor(Math.random() * 4);
    for (var i = 0; i < n; i++) {
      points.push({
        angle: (i / n) * Math.PI * 2,
        r: 0.75 + Math.random() * 0.5
      });
    }
    return points;
  }

  function createRock(size, x, y, vx, vy) {
    return {
      id: state.nextRockId++,
      x: x, y: y, vx: vx, vy: vy,
      rotation: Math.random() * Math.PI * 2,
      rotationSpeed: (Math.random() - 0.5) * 1.6,
      size: size,
      radius: ROCK_SIZES[size].radius,
      shapePoints: makeRockShape()
    };
  }

  // ---- Physics ------------------------------------------------------
  function updateShip(dt) {
    var ship = state.ship;
    if (state.input.left) ship.heading -= TURN_RATE * dt;
    if (state.input.right) ship.heading += TURN_RATE * dt;
    ship.thrusting = state.input.thrust;
    if (ship.thrusting) {
      ship.vx += Math.cos(ship.heading) * THRUST_ACCEL * dt;
      ship.vy += Math.sin(ship.heading) * THRUST_ACCEL * dt;
    }
    var damp = Math.pow(1 - DAMPING_PER_SECOND, dt);
    ship.vx *= damp;
    ship.vy *= damp;
    var speed = Math.sqrt(ship.vx * ship.vx + ship.vy * ship.vy);
    if (speed > MAX_SPEED) {
      var scale = MAX_SPEED / speed;
      ship.vx *= scale;
      ship.vy *= scale;
    }
    ship.x += ship.vx * dt;
    ship.y += ship.vy * dt;
  }

  // ---- Rendering ------------------------------------------------------
  function drawBackground() {
    ctx.fillStyle = COLOR_STAR;
    for (var i = 0; i < stars.length; i++) {
      var s = stars[i];
      ctx.beginPath();
      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  function drawRock(rock) {
    var p = worldToScreen(rock.x, rock.y);
    ctx.save();
    ctx.translate(p.x, p.y);
    ctx.rotate(rock.rotation);
    ctx.beginPath();
    for (var i = 0; i < rock.shapePoints.length; i++) {
      var pt = rock.shapePoints[i];
      var rad = rock.radius * pt.r;
      var px = Math.cos(pt.angle) * rad;
      var py = Math.sin(pt.angle) * rad;
      if (i === 0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
    }
    ctx.closePath();
    ctx.fillStyle = COLOR_ROCK_FILL;
    ctx.fill();
    ctx.strokeStyle = COLOR_ROCK_STROKE;
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.restore();
  }

  // The ship is always drawn at the exact center of the canvas - it's
  // everything else (rocks, later bullets) that gets offset by the
  // ship's world position via worldToScreen(). That's the whole trick.
  function drawShip() {
    var cx = state.canvasCssW / 2, cy = state.canvasCssH / 2;
    var r = state.ship.radius;
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate(state.ship.heading);

    if (state.ship.thrusting) {
      var flameLen = r * (0.8 + Math.random() * 0.5);
      ctx.beginPath();
      ctx.moveTo(-r * 0.35, r * 0.4);
      ctx.lineTo(-r * 0.35 - flameLen, 0);
      ctx.lineTo(-r * 0.35, -r * 0.4);
      ctx.closePath();
      ctx.fillStyle = COLOR_FLAME;
      ctx.fill();
    }

    ctx.beginPath();
    ctx.moveTo(r, 0);
    ctx.lineTo(-r * 0.7, r * 0.7);
    ctx.lineTo(-r * 0.35, 0);
    ctx.lineTo(-r * 0.7, -r * 0.7);
    ctx.closePath();
    ctx.strokeStyle = COLOR_SHIP;
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.restore();
  }

  function render() {
    ctx.clearRect(0, 0, state.canvasCssW, state.canvasCssH);
    drawBackground();
    for (var i = 0; i < state.rocks.length; i++) drawRock(state.rocks[i]);
    drawShip();
  }

  // ---- Game loop ------------------------------------------------------
  var rafId = null;
  function frame(ts) {
    rafId = requestAnimationFrame(frame);
    if (!state.timing.lastFrameTime) { state.timing.lastFrameTime = ts; return; }
    var dt = Math.min((ts - state.timing.lastFrameTime) / 1000, MAX_DT);
    state.timing.lastFrameTime = ts;
    updateShip(dt);
    render();
  }

  window.addEventListener('resize', onResize);
  window.addEventListener('orientationchange', onResize);

  // ---- Init ------------------------------------------------------
  resizeCanvas();
  // A single parked rock (zero velocity) just to make the camera trick
  // visible - lesson 5 replaces this with a real spawning system and
  // gives rocks their own drifting motion.
  state.rocks.push(createRock('large', 130, 0, 0, 0));
  rafId = requestAnimationFrame(frame);
})();
</script>
</body>
