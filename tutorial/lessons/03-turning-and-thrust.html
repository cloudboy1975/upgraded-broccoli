Turning and Thrust

A ship that just sits there isn't a game. This lesson adds keyboard input and real vector physics — turning changes a heading angle, thrust accelerates the ship along that heading, and a bit of drag keeps it from feeling like frictionless deep space (which, ironically, is the physically "correct" version — a little damping just makes it feel controllable).

Reading the keyboard

Input gets translated into a small set of booleans on state.input, decoupled from how it arrived. The physics code below only ever reads state.input and never touches a keyboard event directly — which matters a lot once lesson 11 needs touch input to drive the exact same flags.

JavaScript
function isLeftKey(code) { return code === 'ArrowLeft' || code === 'KeyA'; }
function isRightKey(code) { return code === 'ArrowRight' || code === 'KeyD'; }
function isThrustKey(code) { return code === 'Space' || code === 'ArrowUp' || code === 'KeyW'; }

document.addEventListener('keydown', function (e) {
  if (isThrustKey(e.code)) e.preventDefault(); // stop Space from scrolling the page
  if (isLeftKey(e.code)) state.input.left = true;
  else if (isRightKey(e.code)) state.input.right = true;
  else if (isThrustKey(e.code)) state.input.thrust = true;
});

document.addEventListener('keyup', function (e) {
  if (isLeftKey(e.code)) state.input.left = false;
  else if (isRightKey(e.code)) state.input.right = false;
  else if (isThrustKey(e.code)) state.input.thrust = false;
});

Turning

Turning is just adding to heading, scaled by dt rather than a fixed amount per frame — the same frame-rate independence idea from lesson 1's game loop. At a TURN_RATE of 3.2 radians/second, holding a turn key for about two seconds spins the ship all the way around.

JavaScript
var TURN_RATE = 3.2; // radians/sec

function updateShip(dt) {
  var ship = state.ship;
  if (state.input.left) ship.heading -= TURN_RATE * dt;
  if (state.input.right) ship.heading += TURN_RATE * dt;
  // ...thrust and drag continue below
}

Thrust, drag, and a speed cap

Thrusting accelerates the ship's velocity vector along its current heading — cos(heading)/sin(heading) split the acceleration into x and y components. Every frame, a damping factor bleeds off a percentage of speed; it's written as Math.pow(1 - DAMPING_PER_SECOND, dt) rather than a flat multiply so the drag rate stays correct regardless of frame rate — the same reasoning as the turn rate above, just for an exponential decay instead of a linear one. Finally, speed is clamped to MAX_SPEED so thrust has a ceiling.

JavaScript
var THRUST_ACCEL = 220;
var MAX_SPEED = 260;
var DAMPING_PER_SECOND = 0.5;

function updateShip(dt) {
  var ship = state.ship;
  if (state.input.left) ship.heading -= TURN_RATE * dt;
  if (state.input.right) ship.heading += TURN_RATE * dt;
  ship.thrusting = state.input.thrust;
  if (ship.thrusting) {
    ship.vx += Math.cos(ship.heading) * THRUST_ACCEL * dt;
    ship.vy += Math.sin(ship.heading) * THRUST_ACCEL * dt;
  }
  var damp = Math.pow(1 - DAMPING_PER_SECOND, dt);
  ship.vx *= damp;
  ship.vy *= damp;
  var speed = Math.sqrt(ship.vx * ship.vx + ship.vy * ship.vy);
  if (speed > MAX_SPEED) {
    var scale = MAX_SPEED / speed;
    ship.vx *= scale;
    ship.vy *= scale;
  }
  ship.x += ship.vx * dt;
  ship.y += ship.vy * dt;
}

That's the whole physics model: no gravity, no friction against anything except the artificial damping — simple enough to reason about, expressive enough to feel like an actual spaceship.

The engine flame

One small addition to last lesson's drawShip(): when ship.thrusting is true, draw a flickering triangle behind the ship before the hull itself, so it renders underneath.

JavaScript
var COLOR_FLAME = '#f97316';

// inside drawShip(), after ctx.rotate(state.ship.heading):
if (state.ship.thrusting) {
  var flameLen = r * (0.8 + Math.random() * 0.5);
  ctx.beginPath();
  ctx.moveTo(-r * 0.35, r * 0.4);
  ctx.lineTo(-r * 0.35 - flameLen, 0);
  ctx.lineTo(-r * 0.35, -r * 0.4);
  ctx.closePath();
  ctx.fillStyle = COLOR_FLAME;
  ctx.fill();
}

Randomizing the flame's length by a small amount every frame it's drawn is what makes it read as fire rather than a static shape.

<script>
(function () {
  "use strict";

  // ---- Constants ----------------------------------------------------
  var SHIP_RADIUS = 12;
  var THRUST_ACCEL = 220;
  var MAX_SPEED = 260;
  var DAMPING_PER_SECOND = 0.5;
  var TURN_RATE = 3.2; // radians/sec
  var MAX_DT = 0.05;

  var COLOR_SHIP = '#e2e8f0';
  var COLOR_FLAME = '#f97316';
  var COLOR_STAR = 'rgba(148, 163, 184, 0.5)';

  // ---- DOM ------------------------------------------------------------
  var canvasWrap = document.getElementById('canvasWrap');
  var canvas = document.getElementById('gameCanvas');
  var ctx = canvas.getContext('2d');

  // ---- State ------------------------------------------------------------
  var state = {
    ship: {
      x: 0, y: 0, vx: 0, vy: 0,
      heading: -Math.PI / 2,
      thrusting: false,
      radius: SHIP_RADIUS
    },
    input: { left: false, right: false, thrust: false },
    canvasCssW: 0, canvasCssH: 0, canvasDpr: 1,
    timing: { lastFrameTime: 0 }
  };
  window.__astDebugState = state;

  var stars = [];

  // ---- Canvas sizing ------------------------------------------------------
  function resizeCanvas() {
    var rect = canvasWrap.getBoundingClientRect();
    var dpr = window.devicePixelRatio || 1;
    var w = Math.round(rect.width);
    var h = Math.round(rect.height);
    if (w === state.canvasCssW && h === state.canvasCssH && dpr === state.canvasDpr) return;
    state.canvasCssW = w;
    state.canvasCssH = h;
    state.canvasDpr = dpr;
    canvas.width = Math.round(w * dpr);
    canvas.height = Math.round(h * dpr);
    canvas.style.width = w + 'px';
    canvas.style.height = h + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    generateStars();
  }

  function generateStars() {
    stars = [];
    var count = Math.round((state.canvasCssW * state.canvasCssH) / 4000);
    for (var i = 0; i < count; i++) {
      stars.push({
        x: Math.random() * state.canvasCssW,
        y: Math.random() * state.canvasCssH,
        r: Math.random() * 1.2 + 0.3
      });
    }
  }

  var resizeTimer = null;
  function onResize() {
    clearTimeout(resizeTimer);
    resizeTimer = setTimeout(resizeCanvas, 150);
  }

  // ---- Input: keyboard ------------------------------------------------------
  function isLeftKey(code) { return code === 'ArrowLeft' || code === 'KeyA'; }
  function isRightKey(code) { return code === 'ArrowRight' || code === 'KeyD'; }
  function isThrustKey(code) { return code === 'Space' || code === 'ArrowUp' || code === 'KeyW'; }

  document.addEventListener('keydown', function (e) {
    if (isThrustKey(e.code)) e.preventDefault();
    if (isLeftKey(e.code)) state.input.left = true;
    else if (isRightKey(e.code)) state.input.right = true;
    else if (isThrustKey(e.code)) state.input.thrust = true;
  });

  document.addEventListener('keyup', function (e) {
    if (isLeftKey(e.code)) state.input.left = false;
    else if (isRightKey(e.code)) state.input.right = false;
    else if (isThrustKey(e.code)) state.input.thrust = false;
  });

  // ---- Physics ------------------------------------------------------
  function updateShip(dt) {
    var ship = state.ship;
    if (state.input.left) ship.heading -= TURN_RATE * dt;
    if (state.input.right) ship.heading += TURN_RATE * dt;
    ship.thrusting = state.input.thrust;
    if (ship.thrusting) {
      ship.vx += Math.cos(ship.heading) * THRUST_ACCEL * dt;
      ship.vy += Math.sin(ship.heading) * THRUST_ACCEL * dt;
    }
    var damp = Math.pow(1 - DAMPING_PER_SECOND, dt);
    ship.vx *= damp;
    ship.vy *= damp;
    var speed = Math.sqrt(ship.vx * ship.vx + ship.vy * ship.vy);
    if (speed > MAX_SPEED) {
      var scale = MAX_SPEED / speed;
      ship.vx *= scale;
      ship.vy *= scale;
    }
    ship.x += ship.vx * dt;
    ship.y += ship.vy * dt;
  }

  // ---- Rendering ------------------------------------------------------
  function drawBackground() {
    ctx.fillStyle = COLOR_STAR;
    for (var i = 0; i < stars.length; i++) {
      var s = stars[i];
      ctx.beginPath();
      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  // The ship is always drawn at the exact center of the canvas, no matter
  // where state.ship.x/y actually are - lesson 4 explains why.
  function drawShip() {
    var cx = state.canvasCssW / 2, cy = state.canvasCssH / 2;
    var r = state.ship.radius;
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate(state.ship.heading);

    if (state.ship.thrusting) {
      var flameLen = r * (0.8 + Math.random() * 0.5);
      ctx.beginPath();
      ctx.moveTo(-r * 0.35, r * 0.4);
      ctx.lineTo(-r * 0.35 - flameLen, 0);
      ctx.lineTo(-r * 0.35, -r * 0.4);
      ctx.closePath();
      ctx.fillStyle = COLOR_FLAME;
      ctx.fill();
    }

    ctx.beginPath();
    ctx.moveTo(r, 0);
    ctx.lineTo(-r * 0.7, r * 0.7);
    ctx.lineTo(-r * 0.35, 0);
    ctx.lineTo(-r * 0.7, -r * 0.7);
    ctx.closePath();
    ctx.strokeStyle = COLOR_SHIP;
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.restore();
  }

  function render() {
    ctx.clearRect(0, 0, state.canvasCssW, state.canvasCssH);
    drawBackground();
    drawShip();
  }

  // ---- Game loop ------------------------------------------------------
  var rafId = null;
  function frame(ts) {
    rafId = requestAnimationFrame(frame);
    if (!state.timing.lastFrameTime) { state.timing.lastFrameTime = ts; return; }
    var dt = Math.min((ts - state.timing.lastFrameTime) / 1000, MAX_DT);
    state.timing.lastFrameTime = ts;
    updateShip(dt);
    render();
  }

  window.addEventListener('resize', onResize);
  window.addEventListener('orientationchange', onResize);

  // ---- Init ------------------------------------------------------
  resizeCanvas();
  rafId = requestAnimationFrame(frame);
})();
</script>
